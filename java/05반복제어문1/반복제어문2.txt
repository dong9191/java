import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int oddCount = 0;
        int evenCount = 0;
        
        while (true) {
            int n = sc.nextInt();
            if (n == 0) {
                break;
            }
            if (n % 2 != 0) {
                oddCount++;
            } else {
                evenCount++;
            }
        }
        
        System.out.println("odd : " + oddCount);
        System.out.println("even : " + evenCount);
        sc.close();
    }
}
